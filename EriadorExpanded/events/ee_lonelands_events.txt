					} else = {
						if = {
							limit = { scope:new_ruler = { has_government = tribal_government } }

						} else = {
							title_province = { set_holding_type = castle_holding }

					}

						}

						add_random_building_construct_effect = yes
						add_random_building_variable_effect = yes
						add_random_building_construct_effect = yes
					}
					generate_coa = yes
				}
				add_gold = colonise_cost_val
				create_title_and_vassal_change = {
					type = swear_fealty
					save_scope_as = change
				}
				becomes_independent = {
					change = scope:change
				}
				resolve_title_and_vassal_change = scope:change
				
				if = {
					limit = { scope:new_ruler = { is_elf = yes } }
					change_government = elven_government
				} else_if = {
					limit = { scope:new_ruler = { is_orc = yes } }

					change_government = hobbit_government
				} else = {
					change_government = feudal_government
				}
				
				
			}
			show_as_tooltip = {
				get_title = scope:target1
			}
			
		}
