lonelands = { # Earthquake (Land)
	situation_group_type = major
	is_unique = yes
	
	illustration = "gfx/interface/illustrations/event_scenes/lotr_lonelands.dds"

	##################################################
	# Regions
	##################################################
	sub_regions = {
		eriador = {
			icon = "gfx/interface/icons/situation_types/natural_disaster_earthquake.dds"
			map_color = { 1 0.2 0.2 }
			geographical_regions = { custom_eriador middleearth_forodwaith_forochel middleearth_west_enedhwaith custom_lowlands_angmar }
		}
		
	}

	##################################################
	# On Actions
	##################################################

	on_start = {
		#every_situation_participant = {

		#	set_variable = {
		#		name = colonization_power

		#colonize_power_tick = yes
		#every_situation_county = {
		#	add_county_modifier = block_settlement_ability
		#}
	}

	on_end = {
		
	}

	on_monthly = {
		colonize_power_tick = yes
		every_situation_participant = {
			limit = {
				is_ai = yes
			}
			trigger_event = ee_lonelands.001
		}
		every_situation_participant = {
			limit = {
				var:colonization_power > 100
			}
			save_scope_as = automated_ruler
			if = {
				limit = {
					has_character_flag = lonelands_automation
				}
				while = {
					limit = {
						var:colonization_power > 100
					}
					random_sub_realm_county = {
						limit = {
							any_neighboring_county = {
								any_county_province = {
									has_holding_type = wastelands_holding
								}
								any_county_situation = {
									this = situation:lonelands
								}
							}
						}
						random_neighboring_county = {
							limit = {
								any_county_province = {
									has_holding_type = wastelands_holding
								}
								any_county_situation = {
									this = situation:lonelands
								}
								
							}
							save_scope_as = target3
						}
					}
					random_sub_realm_county = {
						limit = {
							any_county_province = {
								has_holding_type = settlement_holding
							}
						}
						save_scope_as = target5
					}
					random_list = {
						5 = {
							trigger = {
								scope:target3 = {
									exists = yes 
								}
							}
							
							scope:target3 = {
								save_scope_as = target_county
								ai_colonisation_effect = { WASTELANDS = scope:target_county ROOT_SCOPE = scope:automated_ruler }
								change_development_level = { 1 3 }
										
										
								if = {
									limit = { scope:automated_ruler = { is_elf = yes } }
									title_province = { set_holding_type = elven_holding }
								} else_if = {
									limit = { title_province = { terrain = halls } }
									title_province = { set_holding_type = dwarven_holding }
								} else_if = {
									limit = { scope:automated_ruler = { is_hobbit = yes } }
									title_province = { set_holding_type = city_holding }
								} else = {
									if = {
										limit = { scope:automated_ruler = { has_government = tribal_government } }
										title_province = { set_holding_type = tribal_holding }
									} else = {
										title_province = { set_holding_type = castle_holding }
									}
								}		
							}
							add_gold = colonise_cost_val
						}
						1 = {
							random_sub_realm_county = {
								limit = {
									NOT = { 
										any_county_province = {
											has_holding_type = settlement_holding
										}
									}
								}
								change_development_progress_with_overflow = 100
							}
						}
						50 = {
							trigger = {
								scope:target5 = {
									exists = yes 
								}
							}
							scope:target5 = {
								save_scope_as = target_county
								ai_colonisation_effect = { WASTELANDS = scope:target_county ROOT_SCOPE = scope:automated_ruler }
								change_development_level = { 1 3 }
										
										
								if = {
									limit = { scope:automated_ruler = { is_elf = yes } }
									title_province = { set_holding_type = elven_holding }
								} else_if = {
									limit = { title_province = { terrain = halls } }
									title_province = { set_holding_type = dwarven_holding }
								} else_if = {
									limit = { scope:automated_ruler = { is_hobbit = yes } }
									title_province = { set_holding_type = city_holding }
								} else = {
									if = {
										limit = { scope:automated_ruler = { has_government = tribal_government } }
										title_province = { set_holding_type = tribal_holding }
									} else = {
										title_province = { set_holding_type = castle_holding }
									}
								}		
							}
							add_gold = colonise_cost_val
						}
					}
					change_variable = { 
						name = colonization_power 
						add = -100
					}
				}
			}
			
			else = {
				trigger_event = ee_lonelands.002
			}
		}
	}
