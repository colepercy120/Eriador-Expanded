		is_imprisoned = no
		is_lotr_adult = yes
		is_at_war = no
	}
	
	cost = { gold = { value = 100 } }
	
	effect = {
		title:d_shire = {
			set_de_jure_liege_title = title:k_shire
			
		}
		dynasty = { add_dynasty_prestige = 300 }
		trigger_event = {
			id = ee_hobbit.0100
		}
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:decision_shire_independence
		}
	}

decision_reclaim_eriador = {
	title = decision_reclaim_eriador_hobbit
	desc = decision_reclaim_eriador_hobbit_desc
	confirm_text = decision_reclaim_eriador_hobbit_confirm
	selection_tooltip = decision_reclaim_eriador_hobbit_tooltip
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_shire_01.dds"
	}
	

	# AI
	ai_goal = yes
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 100
	}
	
	# IS SHOWN
	decision_group_type = realms_in_exile_major
	
	is_shown = {
		is_hobbit = yes
		OR = {
			
			capital_province = {
				geographical_region = custom_eriador
			}
		
		}
		
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:decision_reclaim_eriador
			}		
		}
	}
	
	cost = {
		prestige = 10000
		piety = 5000
	}
	
	# IS VALID
	is_valid = {
		legitimacy_level >= 5
		prestige_level >= 5
		piety_level >= 5
		
		completely_controls_region = custom_eriador
		
		
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_available_adult = yes
		is_at_war = no
	}

	# EFFECT
	effect = {
		
		
		
		#Create the title and make it primary
		create_title_and_vassal_change = {
			type = created
			save_scope_as = title_change
			add_claim_on_loss = yes
		}
		
		title:e_eriador = {
			change_title_holder = {
				holder = root
				change = scope:title_change
			}
		}
		resolve_title_and_vassal_change = scope:title_change
		
		
		if = {
			limit = { highest_held_title_tier < tier_empire }
			set_primary_title_to = title:e_eriador
		}
		
		
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:decision_reclaim_eriador
		}
		
		custom_tooltip = {
			text = decision_reclaim_eriador_effect
			every_kingdom = {
				limit = {
					title_capital_county = {
						any_county_province = {
							OR = {
								geographical_region = custom_eriador
								geographical_region = middleearth_forodwaith
							}
						}
					}
				}
				set_de_jure_liege_title = title:e_eriador
			}
		}

		

		trigger_event = {
			id = ee_hobbit.0101
		}
	}
}
